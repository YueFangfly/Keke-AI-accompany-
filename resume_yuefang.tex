% Project 3: Evacuation
\projectHeading
  {Crowd Evacuation Modelling -- PDE vs.\ Agent-Based Simulation}
  {Python, PDE, SDE, Monte Carlo, Queuing Theory}
  {2026}

\begin{itemize}[leftmargin=1.2em, itemsep=1pt, parsep=0pt, topsep=2pt]
  \item Developed two complementary mathematical models for \textbf{emergency evacuation} of a 300-person venue, comparing \textbf{macroscopic (PDE)} and \textbf{microscopic (agent-based)} approaches against UK fire safety regulations (150s evacuation limit)
  \item Implemented a \textbf{macroscopic PDE model} solving the \textbf{Hughes continuity equation} via finite volume discretization on a 15$\times$30 grid, with density-dependent velocity ($v(\rho) = v_{\max}(1 - \rho/\rho_{\max})$), BFS-computed direction fields, and CFL-stable explicit Euler time integration ($\Delta t = 0.1$s)
  \item Built a \textbf{microscopic agent-based model} using \textbf{Stochastic Differential Equations} (SDE) with Brownian motion for behavioural randomness, discretized via the \textbf{Euler-Maruyama method}, and \textbf{$M/E_r/2$ queuing} at exits with Erlang-$k$ service times
  \item Modelled \textbf{population heterogeneity} (mobility-impaired agents at $v_{\max} = 0.75$ m/s vs.\ 1.5 m/s) and analysed impact on bottleneck formation; ran \textbf{100 Monte Carlo simulations} for statistical robustness
  \item Performed \textbf{exit placement optimization}: centre-aligned repositioning reduced evacuation time by 35\% (from 98.6s to 75.5s), validated through \textbf{bottleneck heatmap analysis} identifying congestion zones
  \item Delivered actionable \textbf{safety recommendations} for venue design, demonstrating that combining macroscopic and microscopic modelling provides more robust guidance than either approach alone
\end{itemize}
